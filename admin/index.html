Kích Hoạt Quyền & Bảng Mới Trên Supabase

Vào Supabase Dashboard ➔ SQL Editor ➔ Dán toàn bộ câu lệnh bên dưới và bấm Run:

-- 1. Cho phép cập nhật / xóa đơn hàng trực tiếp từ Admin
CREATE POLICY IF NOT EXISTS "anon_update_orders" ON orders FOR UPDATE TO anon USING (true) WITH CHECK (true);
CREATE POLICY IF NOT EXISTS "anon_delete_orders" ON orders FOR DELETE TO anon USING (true);

-- 2. Tạo bảng products trên Cloud Supabase
CREATE TABLE IF NOT EXISTS products (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name TEXT NOT NULL,
    type TEXT NOT NULL DEFAULT 'physical',
    price BIGINT NOT NULL,
    description TEXT,
    stock INTEGER DEFAULT 100,
    created_at TIMESTAMPTZ DEFAULT NOW()
);
ALTER TABLE products ENABLE ROW LEVEL SECURITY;
CREATE POLICY IF NOT EXISTS "anon_all_products" ON products FOR ALL TO anon USING (true) WITH CHECK (true);

-- Thêm 3 gói sản phẩm có sẵn
INSERT INTO products (name, type, price, description, stock) VALUES

('Nano Growth Habit EX - Gói Toàn Diện (3 hộp)', 'physical', 3390000, '3 hộp 360 viên (Mua 2 Tặng 1), dùng ~6 tháng.', 100)
ON CONFLICT DO NOTHING;

-- 3. Tạo bảng customers trên Cloud Supabase
CREATE TABLE IF NOT EXISTS customers (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name TEXT NOT NULL,
    phone TEXT NOT NULL UNIQUE,
    zalo TEXT,
    note TEXT,
    registered_at TIMESTAMPTZ DEFAULT NOW()
);
ALTER TABLE customers ENABLE ROW LEVEL SECURITY;
CREATE POLICY IF NOT EXISTS "anon_all_customers" ON customers FOR ALL TO anon USING (true) WITH CHECK (true);
